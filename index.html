def simulate_marketing_campaign(budget: float, cpc: float, conversion_rate: float, aov: float) -> dict:
    """
    Simulador de Campanha de Marketing Digital
    Desenvolvido por Daniel Marques
    """
    # 1. Tráfego gerado a partir do orçamento e custo por clique
    clicks = int(budget / cpc) if cpc > 0 else 0
    
    # 2. Número de conversões baseado na taxa de conversão (CR)
    conversions = int(clicks * (conversion_rate / 100.0))
    
    # 3. Retorno financeiro e métricas-chave
    revenue = conversions * aov
    net_profit = revenue - budget
    roas = (revenue / budget) if budget > 0 else 0.0
    cac = (budget / conversions) if conversions > 0 else 0.0
    
    return {
        "budget": budget,
        "clicks": clicks,
        "conversions": conversions,
        "revenue": round(revenue, 2),
        "net_profit": round(net_profit, 2),
        "roas": round(roas, 2),
        "cac": round(cac, 2)
    }
